# RESETING GITHUB TO A PREVIOUS COMMIT

# 1. Download latest remote history WITHOUT merging
git fetch origin

# 2. See recent commit history
git log --oneline --graph --decorate --all -20

# 3. Identify the LAST GOOD commit hash
# Example:
# 8f56f5f  good commit
# b2c72cb  bad commit

# 4. Restore repository to the good commit
git reset --hard GOOD_COMMIT_HASH

# Example:
git reset --hard 8f56f5f

# 5. Overwrite GitHub branch with restored version
git push --force origin main


# RESOLVING CONFLICTS
